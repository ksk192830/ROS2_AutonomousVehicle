cmake_minimum_required(VERSION 3.8)
project(exhibition_sim_plugins)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(std_msgs REQUIRED)
find_package(gazebo_msgs REQUIRED)
add_library(exhibition_camera_filter SHARED src/camera_filter.cpp)
target_compile_features(exhibition_camera_filter PRIVATE cxx_std_17)
target_include_directories(exhibition_camera_filter PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(exhibition_camera_filter PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(exhibition_camera_filter ${GAZEBO_LIBRARIES})
install(TARGETS exhibition_camera_filter LIBRARY DESTINATION lib)
add_library(exhibition_camera_rate SHARED src/camera_rate.cpp)
target_compile_features(exhibition_camera_rate PRIVATE cxx_std_17)
target_include_directories(exhibition_camera_rate PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(exhibition_camera_rate PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(exhibition_camera_rate ${GAZEBO_LIBRARIES})
ament_target_dependencies(exhibition_camera_rate gazebo_ros std_msgs)
install(TARGETS exhibition_camera_rate LIBRARY DESTINATION lib)
add_library(exhibition_compact_state SHARED src/compact_state.cpp)
target_compile_features(exhibition_compact_state PRIVATE cxx_std_17)
target_include_directories(exhibition_compact_state PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(exhibition_compact_state PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(exhibition_compact_state ${GAZEBO_LIBRARIES})
ament_target_dependencies(exhibition_compact_state gazebo_ros gazebo_msgs)
install(TARGETS exhibition_compact_state LIBRARY DESTINATION lib)
ament_package()
